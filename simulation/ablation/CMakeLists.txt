cmake_minimum_required(VERSION 3.20)
project(drone_ablation LANGUAGES CXX)
set(COMPANION_BUILD_RUNNER OFF CACHE BOOL "" FORCE)
set(COMPANION_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(../companion companion)
find_package(nlohmann_json REQUIRED)
add_executable(algorithm_ablation algorithm_ablation.cpp)
target_link_libraries(algorithm_ablation PRIVATE companion_core nlohmann_json::nlohmann_json)
target_compile_features(algorithm_ablation PRIVATE cxx_std_17)
target_compile_options(algorithm_ablation PRIVATE -Wall -Wextra -Wpedantic)
