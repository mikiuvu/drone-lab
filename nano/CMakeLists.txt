cmake_minimum_required(VERSION 3.16)
project(drone_nano C)
set(CMAKE_C_STANDARD 11)
add_executable(drone-nano main.c platform.c ../vendor/stm32-vl53l5cx/modules/vl53l5cx_api.c)
target_include_directories(drone-nano PRIVATE . ../vendor/stm32-vl53l5cx/modules)
target_include_directories(drone-nano SYSTEM PRIVATE ../vendor/mavlink)
target_compile_options(drone-nano PRIVATE -Wall -Wextra -O2 -g)
target_link_libraries(drone-nano PRIVATE m)
if(STATIC_BUILD)
  target_link_options(drone-nano PRIVATE -static)
endif()
